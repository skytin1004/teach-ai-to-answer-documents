# តារាងត្រួតពិនិត្យការបោះពុម្ពផ្សាយ

ប្រើតារាងត្រួតពិនិត្យនេះមុនពេលធ្វើការបញ្ជាទុកឬផ្ទុកការអាប់ដេតសាធារណៈ។

## សុវត្ថិភាព

- ធានាថាគ្មានគ្រាប់ចូល API, តូកិន, ពាក្យសម្ងាត់ ឬចំណុចចូលឯកជនណា ទេដែលបានសរសេរចូលក្នុងឯកសារ Markdown, សៀវភៅកំណត់ត្រា, ទិន្នន័យគំរូ ឬស្ក្រីប។ 
- រក្សាទុកសម្ងាត់ក្នុងអថេរបរិបទបរិស្ថាន ឬអត្តសញ្ញាណគ្រប់គ្រង មិនមែនក្នុងឯកសារដែលបានបញ្ជាទុកទេ។
- មិនដែលបញ្ជាទុកឯកសារ `.env` ឬឯកសារចេញពីការប្រតិបត្តិសៀវភៅកំណត់ត្រា។ 
- រក្សា `.env.example` ជាតែគំរូដើម្បីបំពេញតែប៉ុណ្ណោះ។

## ការត្រួតពិនិត្យ

រត់ស្គ្រីបត្រួតពិនិត្យវេទិកាគម្រោង:

```powershell
python scripts\verify_notebooks.py
```

រត់ការប្រតិបត្តិការសៀវភៅកំណត់ត្រានៅក្នុងស្រុកដែលមានសុវត្ថិភាពរួចមុនពេលបោះពុម្ពផ្សាយការផ្លាស់ប្ដូរកម្មវិធី៖

```powershell
python scripts\verify_notebooks.py --execute
```

ការត្រួតពិនិត្យដែលរំពឹងទុក៖

- តំណភ្ជាប់ Markdown ក្នុងស្រុកឆ្លងកាត់
- ការផ្ទៀងផ្ទាត់ JSON នៃសៀវភៅកំណត់ត្រាឆ្លងកាត់
- សៀវភៅកំណត់ត្រា មិនមានការផ្ទុកលទ្ធផលដែលបានរក្សាទុកឬចំនួនការប្រតិបត្តិការណ៍
- ការត្រួតពិនិត្យលំនាំសម្ងាត់មានហានិភ័យខ្ពស់ឆ្លងកាត់
- សៀវភៅកំណត់ត្រាសាធារណៈ អាចបំពេញជាស្រុក
- សារារងក្រោម `drafts/` ត្រូវបានរំលងដោយមានចេតនា

## ការត្រួតពិនិត្យឡើងវិញ

- ធានាថាតំណអត្ថបទ README មានការដែលបង្ហាញទៅឯកសារដែលចង់បាន។
- ធានាថាអត្ថបទនិមួយៗមានលទ្ធភាពចាកចេញទៅកាន់គេហទំព័រនិងតំណសៀវភៅកំណត់ត្រាពាក់ព័ន្ធគ្នា។
- ធានាថារូបត្រា មិនត្រូវបានភ្ជាប់ពីវិញ្ញាបនបត្រសាធារណៈទេ លើកលែងតែរួចរាល់សម្រាប់បោះពុម្ពផ្សាយ។
- ធានាថាលំនាំបែបបទបញ្ហា និងការទាញពាក្យ GitHub ទំនើបស្របនឹងដំណើរការគម្រោង។
- ធានាថាលទ្ធផលត្រួតពិនិត្យក្នុងអត្ថបទត្រូវគ្នានឹងលទ្ធផលចុងក្រោយនៃសៀវភៅកំណត់ត្រា។
- ធានាថាការងារសកម្មភាព GitHub ត្រូវរំពឹងអោយរត់បន្ទាប់ពីការផ្ទុកឡើង។
- ធានាថា `CHANGELOG.md` បង្ហាញពីការអាប់ដេតដែលកំពុងបោះពុម្ពផ្សាយ។
- ធានាថា `CONTRIBUTING.md` នៅតែស្របនឹងដំណើរការគម្រោង។

## Git

- ពិនិត្យ `git status --short --branch`។
- ពិនិត្យ `git diff --stat`។
- បញ្ជាទុក និងផ្ទុកឡើងតែពេលដែលបានរៀបចំយ៉ាងច្បាស់។
